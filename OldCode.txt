void game::magic_to_file() {
    stringstream ss_bishop;
    stringstream ss_rook;
    ofstream fs_bishop("bishop_magics.txt");
    ofstream fs_rook("rook_magics.txt");
    ss_bishop << "{" << '\n' << std::hex;
    ss_rook << "{" << '\n' << std::hex;
      
    for(Squares square{ a8 }; square < no_square; ++square) {
      const u64 magic_bishop = find_magic_number(square, bishop_relevant_bits[square], sliders::bishop);
      ss_bishop <<"0x" << magic_bishop << ",";
    if (square % 4 == 0 && square != 0)  
      ss_bishop << '\n';
  }

  for(Squares square{ a8 }; square < no_square; ++square) {
    const u64 magic_rook = find_magic_number(square, rook_relevant_bits[square], sliders::rook);
    ss_rook <<"0x" << magic_rook << ",";
    if (square % 4 == 0 && square != 0)  
      ss_rook << '\n';  
  }
  ss_bishop << "}\n";
  ss_rook << "}\n";
  fs_bishop << ss_bishop.str();
  fs_bishop.close();
  fs_rook << ss_rook.str();
  fs_rook.close();
}
